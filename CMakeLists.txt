cmake_minimum_required(VERSION 3.20)
project(note4c_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NOTE4C_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(NOTE4C_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(components/note4c_core)
add_executable(note4c_sim examples/simulator.cpp)
target_link_libraries(note4c_sim PRIVATE note4c_core)
add_executable(note4c_cal_preview examples/calendar_preview.cpp)
target_link_libraries(note4c_cal_preview PRIVATE note4c_core)
enable_testing()
add_executable(note4c_tests tests/core_tests.cpp)
target_link_libraries(note4c_tests PRIVATE note4c_core)
add_test(NAME note4c_core COMMAND note4c_tests)
add_executable(note4c_calendar_tests tests/calendar_tests.cpp)
target_link_libraries(note4c_calendar_tests PRIVATE note4c_core)
add_test(NAME note4c_calendar COMMAND note4c_calendar_tests)
